# ---------------------------------------------------------------------------
# Version, derived from git so it moves on its own for every commit.  Hand
# editing a number is exactly the step that gets forgotten, and then an old
# vpk is indistinguishable from a new one -- which has already cost this
# project several confusing test runs.
#
# APP_VER is a fixed XX.XX, so the commit count becomes major.minor with the
# major starting at 01: every build sorts above the hand-set 00.51 that came
# before, and the pair only repeats after ten thousand commits.
execute_process(
  COMMAND git rev-list --count HEAD
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE GIT_COMMIT_COUNT
  OUTPUT_STRIP_TRAILING_WHITESPACE
  ERROR_QUIET)
execute_process(
  COMMAND git rev-parse --short HEAD
  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE GIT_COMMIT_SHA
  OUTPUT_STRIP_TRAILING_WHITESPACE
  ERROR_QUIET)

if(NOT GIT_COMMIT_COUNT)
  # Building from a tarball rather than a checkout.
  set(GIT_COMMIT_COUNT 0)
  set(GIT_COMMIT_SHA "unknown")
endif()

math(EXPR ONS_VERSION_MAJOR "1 + ${GIT_COMMIT_COUNT} / 100")
math(EXPR ONS_VERSION_MINOR "${GIT_COMMIT_COUNT} % 100")
if(ONS_VERSION_MAJOR LESS 10)
  set(ONS_VERSION_MAJOR "0${ONS_VERSION_MAJOR}")
endif()
if(ONS_VERSION_MINOR LESS 10)
  set(ONS_VERSION_MINOR "0${ONS_VERSION_MINOR}")
endif()
set(ONS_VERSION "${ONS_VERSION_MAJOR}.${ONS_VERSION_MINOR}")

string(TIMESTAMP ONS_BUILD_DATE "%Y-%m-%d" UTC)

# Both binaries print this at startup, so a log says which build produced it
# without anyone having to remember what they installed.  Written before the
# subdirectories are added, since both compile against it.
file(WRITE ${CMAKE_BINARY_DIR}/build_version.h
"/* Generated by cmake -- do not edit. */\n"
"#ifndef ONS_BUILD_VERSION_H\n"
"#define ONS_BUILD_VERSION_H\n"
"#define ONS_BUILD_VERSION \"${ONS_VERSION}\"\n"
"#define ONS_BUILD_COMMIT  \"${GIT_COMMIT_SHA}\"\n"
"#define ONS_BUILD_DATE    \"${ONS_BUILD_DATE}\"\n"
"#define ONS_BUILD_STRING  \"ONS Easy Setup ${ONS_VERSION} (${GIT_COMMIT_SHA}, ${ONS_BUILD_DATE})\"\n"
"#endif\n")
message(STATUS "ONS Easy Setup version ${ONS_VERSION} (${GIT_COMMIT_SHA})")
